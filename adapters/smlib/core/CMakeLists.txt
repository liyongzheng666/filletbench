cmake_minimum_required(VERSION 3.20)
project(filletbench_smlib_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED SMLIB_ROOT AND DEFINED ENV{SMLIB_ROOT})
  set(SMLIB_ROOT "$ENV{SMLIB_ROOT}")
endif()
if(NOT EXISTS "${SMLIB_ROOT}/source/SMLib/inc/SmConfig.h")
  message(FATAL_ERROR "Set SMLIB_ROOT to a USD-BRep source checkout")
endif()
file(GLOB CORE CONFIGURE_DEPENDS "${SMLIB_ROOT}/source/SMLib/src/*.cpp")
file(GLOB API CONFIGURE_DEPENDS "${SMLIB_ROOT}/source/SM_API/src/*.cpp")
# This optional importer/exporter depends on USD and is unused by this adapter.
list(FILTER API EXCLUDE REGEX "SmApiImportExport\\.cpp$")
add_library(smlib_filletbench SHARED ${CORE} ${API})
target_include_directories(smlib_filletbench PRIVATE
  "${SMLIB_ROOT}/source/SMLib/inc" "${SMLIB_ROOT}/source/SM_API/inc")
target_compile_definitions(smlib_filletbench PRIVATE NDEBUG NOMINMAX SM_API_EXPORTS SM_GRAPHICS_CALLBACKS=1)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(smlib_filletbench PRIVATE -Wno-overloaded-virtual -Wno-deprecated-declarations)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  target_compile_options(smlib_filletbench PRIVATE -Wno-inconsistent-missing-override)
endif()
